General Notes:

\(\mu_i = 2\arctan(\beta_0 + X_i^T\beta)\) : link function

\(Y_i \mid X_i \sim \mathrm{VM}(\mu_i, \kappa)\) : Von Meises distribution

Concentration Parameter \(\kappa\):

0 - uniform dist around circle

0.5 - very low concentration

1 - low concentration

5 - moderate concentration

10 - high concentration

20 - very high concentration

Model (10/30)

set.seed(1)

n <- 5000
x <- data.frame(x1 = rnorm(n), x2 = rnorm(n), x3 = rnorm(n))

We want to answer the question: How does kappa (concentration) and the location of the splitting point affect the accuracy of linear regression when analyzing circular data, and under what conditions is circular regression necessary?

^ doing this by looking at the \(\mu\)s

^ by rotating the responses, we are changing the splitting point.

set.seed(1)

# parameters
n <- 500
beta0 <- 2
beta <- 2

# vary kappas
kappas <- c(0.3, 0.5, 0.75, 1)

# different rotations
rotation <- 0

# store estimates
beta_linear_0 <- beta_circular_0 <- list()

nsim <- 100

for (i in 1:length(kappas)) {
  
  k <- kappas[i]
  beta_linear_0[[i]] <- beta_circular_0[[i]] <- list()
  
  for (j in 1:nsim) {
    print(j)
    
    x <- rnorm(n)
    mu <- beta0 + 2*atan(beta*x)
    
    # generate original circular responses
    y_original <- sapply(mu, function(m) {
      as.numeric(rvonmises(1, mu = circular(m), kappa = k))})
    
    # rotate observations and split at 0
    y <- (y_original - rotation) %% (2*pi)
    
    # fit linear regression
    mod1 <- lm(y ~ x)
    
    beta_linear_0[[i]][j] <- coef(mod1)[1] + mean(x) * coef(mod1)[2]
    
    # fit circular regression
    mod2 <- lm.circular(y = circular(y), x = matrix(x, ncol = 1), type = "c-l", init = 1)
    
    beta_circular_0[[i]][j] <- mod2$mu + 2*atan(mean(x) * mod2$coefficients[1])
  }
}
